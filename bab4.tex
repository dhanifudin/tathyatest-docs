\noindent
\textbf{Keterangan Persentase:}
\begin{itemize}
    \item Biaya Sewa Alat: 15\% dari sub total (maksimal 15\%)
    \item Biaya Bahan: 40\% dari sub total (minimal 40\%)
    \item Biaya Perjalanan: 20\% dari sub total (maksimal 20\%)
    \item Biaya Lain-lain: 25\% dari sub total (publikasi, komunikasi, dokumentasi, pelaporan, honorarium pembantu peneliti)
\end{itemize}

\noindent
\textbf{Penjelasan Kegiatan:}
\begin{enumerate}
    \item \textbf{Persiapan dan Studi Literatur:} Pengumpulan referensi, kajian penelitian terdahulu, dan persiapan tools yang dibutuhkan.
    \item \textbf{Desain Sistem TathyaTest:} Perancangan arsitektur sistem, desain database, dan alur kerja TathyaTest.
    \item \textbf{Pengembangan Prototipe CRUD:} Implementasi fitur dasar untuk crawling dan test case generation aplikasi CRUD sederhana.
    \item \textbf{Pengujian Prototipe Awal:} Testing dan debugging prototipe awal, pengumpulan umpan balik pengguna.
    \item \textbf{Pengembangan Fitur Lanjutan:} Penambahan fitur untuk form dinamis, modul interaktif, dan penyempurnaan algoritma.
    \item \textbf{Integrasi Playwright:} Implementasi eksekusi test case otomatis dengan Playwright untuk cross-browser testing.
    \item \textbf{Pengujian dengan Dataset Internal:} Testing TathyaTest dengan aplikasi Kompen, Jayanti, dan Whistleblower.
    \item \textbf{Analisis Data dan Evaluasi:} Analisis hasil pengujian, evaluasi metrik kinerja, dan validasi dengan pakar.
    \item \textbf{Penyusunan Laporan:} Penulisan laporan akhir penelitian dan dokumentasi lengkap TathyaTest.
    \item \textbf{Publikasi Artikel:} Penulisan dan pengiriman artikel ke jurnal terakreditasi Sinta 3.
    \item \textbf{Pendaftaran HKI:} Pendaftaran Hak Cipta TathyaTest ke DJKI Kementerian Hukum dan HAM RI.
\end{enumerate}
